#Busquedas booleanas must con 2 condiciones
GET /productos/_search
{
  "query": {
    "bool": {
      "must": [
        { "match": {
          "name": "glass"
        }},
        {"range": {
          "price": {
            "lte": 150
          }
        }}
      ]
    }
  }
}

#Busqueda booleana must, must not, should y filter
GET /productos/_search
{
  "query": {
    "bool": {
    "must": [
    {"match": {
      "name": "glass"
    }}
  ],
  "must_not": [
    {"match": {
      "name": "meal"
    }}
  ],
  "should": [
    {"range": {
      "price": {
        "gte": 10,
        "lte": 150
      }
    }}
  ],
  "filter": [
    {"range": {
      "price": {
        "lte": 100
      }
    }}
  ]
  }}
}
