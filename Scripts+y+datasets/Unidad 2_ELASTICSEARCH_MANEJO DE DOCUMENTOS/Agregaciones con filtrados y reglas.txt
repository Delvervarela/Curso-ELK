#Crear agregacion con un filtrado previo
GET /pedidos/_search
{
  "size": 0,
  "aggs": {
    "valor_bajo": {
      "filter": {
        "range": {
          "total_importe": {
            "lte": 50
          }
        }
      },
      "aggs": {
        "promedio_importe": {
          "avg": {
            "field": "total_importe"
        }
      }
    }
  }
}
}

      "filters": {

            }

          }
        }
      },
      "aggs": {

    }
  }
